AQARIONFinite.duality {n : ℕ} (T : Fin n → Fin n) (v g : Fin n → ℚ) :
  (AQARIONFinite.pairing v fun x => g (T x)) = AQARIONFinite.pairing (AQARIONFinite.push T v) g

  AQARIONFinite.pairing (AQARIONFinite.scaleWeight a v) g = a * AQARIONFinite.pairing v g
